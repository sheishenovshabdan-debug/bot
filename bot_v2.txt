def now() -> datetime:
    """Текущее время в часовом поясе салона (а не в поясе сервера)."""
    return datetime.now(TZ)


def today_str() -> str:
    return now().date().isoformat()


def iso_now() -> str:
    return now().isoformat(timespec="seconds")


def slot_times() -> list[str]:
    return [f"{h:02d}:00" for h in WORK_HOURS]


def slot_dt(day: str, t: str) -> datetime:
    d = date.fromisoformat(day)
    hh, mm = (int(x) for x in t.split(":"))
    return datetime(d.year, d.month, d.day, hh, mm, tzinfo=TZ)


def fmt_day(day: str) -> str:
    d = date.fromisoformat(day)
    return f"{d:%d.%m} ({WEEKDAYS[d.weekday()]})"


def day_human(day: str) -> str:
    delta = (date.fromisoformat(day) - now().date()).days
    if delta == 0:
        return "сегодня"
    if delta == 1:
        return "завтра"
    return fmt_day(day)


def esc(value: object) -> str:
    return html.escape(str(value or ""), quote=False)


def fmt_phone(phone: str) -> str:
    digits = re.sub(r"\D", "", phone or "")
    if len(digits) == 11 and digits[0] in "78":
        return f"+7 ({digits[1:4]}) {digits[4:7]}-{digits[7:9]}-{digits[9:11]}"
    return phone or ""


def normalize_phone(raw: str) -> str | None:
    digits = re.sub(r"\D", "", raw or "")
    if not digits:
        return None
    if len(digits) == 11 and digits[0] == "8":
        digits = "7" + digits[1:]
    elif len(digits) == 10:
        digits = "7" + digits
    if not 10 <= len(digits) <= 15:
        return None
    return "+" + digits


def clean_name(raw: str) -> str | None:
    value = " ".join((raw or "").split())
    if not 2 <= len(value) <= 50:
        return None
    if len(re.findall(r"[A-Za-zА-Яа-яЁё]", value)) < 2:
        return None
    return value


def clean_model(raw: str) -> str | None:
    value = " ".join((raw or "").split())
    if not 2 <= len(value) <= 60:
        return None
    if not re.search(r"[A-Za-zА-Яа-яЁё0-9]", value):
        return None
    return value


def is_admin(user) -> bool:
    return bool(ADMIN_ID) and getattr(user, "id", None) == ADMIN_ID


def is_bookable_day(day: str) -> bool:
    try:
        d = date.fromisoformat(day)
    except ValueError:
        return False
    today = now().date()
    if not today <= d <= today + timedelta(days=DAYS_AHEAD - 1):
        return False
    return d.weekday() not in DAYS_OFF

def _connect() -> sqlite3.Connection:
    conn = sqlite3.connect(DB_PATH, timeout=15)
    conn.row_factory = sqlite3.Row
    conn.execute("PRAGMA journal_mode=WAL")   # меньше блокировок при параллельных запросах
    conn.execute("PRAGMA busy_timeout=5000")
    return conn


@contextmanager
def db():
    """Соединение, которое всегда закрывается (в старом коде утекали дескрипторы)."""
    conn = _connect()
    try:
        yield conn
        conn.commit()
    except Exception:
        conn.rollback()
        raise
    finally:
        conn.close()


def _rebuild_legacy_table(conn: sqlite3.Connection) -> None:
    """Старая схема содержала UNIQUE(day, time) — из-за него нельзя было
    отменить запись и занять слот снова. Пересобираем таблицу без него."""
    row = conn.execute(
        "SELECT sql FROM sqlite_master WHERE type='table' AND name='bookings'"
    ).fetchone()
    if not row or not row["sql"] or "UNIQUE" not in row["sql"].upper():
        return

    old_cols = {r["name"] for r in conn.execute("PRAGMA table_info(bookings)")}
    conn.execute("ALTER TABLE bookings RENAME TO bookings_legacy")
    conn.execute(TABLE_BOOKINGS.format(name="bookings"))

    new_cols = [r["name"] for r in conn.execute("PRAGMA table_info(bookings)")]
    source: dict[str, str] = {}
    for col in new_cols:
        if col in old_cols:
            source[col] = col
        elif col == "reminded_day" and "reminded" in old_cols:
            source[col] = "reminded"

    targets = list(source)
    conn.execute(
        f"INSERT INTO bookings ({', '.join(targets)}) "
        f"SELECT {', '.join(source[c] for c in targets)} FROM bookings_legacy"
    )
    conn.execute("DROP TABLE bookings_legacy")
    logging.info("Миграция bookings выполнена: старый UNIQUE(day, time) снят")


def _add_columns(conn: sqlite3.Connection, table: str, columns: dict[str, str]) -> None:
    have = {r["name"] for r in conn.execute(f"PRAGMA table_info({table})")}
    for column, ddl in columns.items():
        if column not in have:
            conn.execute(f"ALTER TABLE {table} ADD COLUMN {column} {ddl}")


def init_db() -> None:
    with db() as conn:
        _rebuild_legacy_table(conn)
        conn.execute(TABLE_BOOKINGS.format(name="bookings"))
        conn.execute(TABLE_USERS)
        conn.execute(TABLE_CLOSURES)
        conn.execute(TABLE_REVIEWS)
        _add_columns(conn, "bookings", {
            "username": "TEXT",
            "price": "TEXT",
            "status": "TEXT NOT NULL DEFAULT 'pending'",
            "reminded_day": "INTEGER NOT NULL DEFAULT 0",
            "reminded_soon": "INTEGER NOT NULL DEFAULT 0",
            "created_at": "TEXT NOT NULL DEFAULT ''",
            "photo_id": "TEXT",
            "ready_at": "TEXT",
            "review_requested": "INTEGER NOT NULL DEFAULT 0",
        })
        _add_columns(conn, "users", {
            "opt_in": "INTEGER NOT NULL DEFAULT 1",
        })
        conn.executescript(INDEXES)


def busy_slots(day: str) -> set[str]:
    placeholders = ",".join("?" for _ in OCCUPYING_STATUSES)
    with db() as conn:
        rows = conn.execute(
            f"SELECT time FROM bookings WHERE day=? AND status IN ({placeholders})",
            (day, *OCCUPYING_STATUSES),
        )
        return {r["time"] for r in rows}


def closed_slots(day: str) -> set[str]:
    """Слоты, закрытые владельцем: выходной, обед, отпуск."""
    with db() as conn:
        rows = conn.execute("SELECT time FROM closures WHERE day=?", (day,)).fetchall()
    if any(r["time"] is None for r in rows):
        return set(slot_times())
    return {r["time"] for r in rows if r["time"]}


def free_slots(day: str, exclude: sqlite3.Row | None = None) -> list[str]:
    """Свободные слоты дня с учётом занятости, закрытий и запаса времени."""
    if not is_bookable_day(day):
        return []
    busy = busy_slots(day)
    if exclude is not None and exclude["day"] == day:
        busy.discard(exclude["time"])
    blocked = busy | closed_slots(day)
    deadline = now() + timedelta(minutes=LEAD_MINUTES)
    return [t for t in slot_times() if t not in blocked and slot_dt(day, t) > deadline]


def upsert_user(user) -> None:
    if user is None:
        return
    with db() as conn:
        conn.execute(
            """INSERT INTO users (user_id, username, first_name, created_at, seen_at)
               VALUES (?, ?, ?, ?, ?)
               ON CONFLICT(user_id) DO UPDATE SET
                   username=excluded.username,
                   first_name=excluded.first_name,
                   seen_at=excluded.seen_at""",
            (user.id, user.username, user.first_name, iso_now(), iso_now()),
        )

# ---------- закрытие дней и часов ----------

def add_closure(day: str, slot: str | None = None, reason: str = "") -> None:
    with db() as conn:
        conn.execute(
            "INSERT OR IGNORE INTO closures (day, time, reason, created_at) VALUES (?,?,?,?)",
            (day, slot, reason, iso_now()))


def remove_closure(day: str, slot: str | None = None) -> int:
    with db() as conn:
        if slot is None:
            cursor = conn.execute("DELETE FROM closures WHERE day=?", (day,))
        else:
            cursor = conn.execute("DELETE FROM closures WHERE day=? AND time=?", (day, slot))
        return cursor.rowcount


def upcoming_closures(days: int = 14) -> list[sqlite3.Row]:
    with db() as conn:
        return conn.execute(
            "SELECT * FROM closures WHERE day BETWEEN ? AND ? ORDER BY day, time",
            (today_str(), (now().date() + timedelta(days=days)).isoformat()),
        ).fetchall()


# ---------- гарантия ----------

def add_months(value: date, months: int) -> date:
    month_index = value.month - 1 + months
    year = value.year + month_index // 12
    month = month_index % 12 + 1
    return date(year, month, min(value.day, calendar.monthrange(year, month)[1]))


def warranty_rows(phone: str) -> list[sqlite3.Row]:
    with db() as conn:
        return conn.execute(
            "SELECT * FROM bookings WHERE phone=? AND ready_at IS NOT NULL "
            "AND status IN ('ready','done') ORDER BY ready_at DESC",
            (phone,),
        ).fetchall()


# ---------- отзывы ----------

def save_review(booking_id: int, user_id: int, rating: int) -> None:
    with db() as conn:
        conn.execute(
            "INSERT INTO reviews (booking_id, user_id, rating, created_at) VALUES (?,?,?,?)",
            (booking_id, user_id, rating, iso_now()))
        conn.execute("UPDATE bookings SET review_requested=1 WHERE id=?", (booking_id,))


def review_stats() -> tuple[float, int]:
    with db() as conn:
        row = conn.execute("SELECT AVG(rating), COUNT(*) FROM reviews").fetchone()
    average = round(float(row[0]), 1) if row[0] is not None else 0.0
    return average, int(row[1])


def recent_reviews(limit: int = 5) -> list[sqlite3.Row]:
    with db() as conn:
        return conn.execute(
            "SELECT r.*, b.name, b.service FROM reviews r "
            "JOIN bookings b ON b.id=r.booking_id ORDER BY r.id DESC LIMIT ?",
            (limit,),
        ).fetchall()

def _worksheet():

    from google.oauth2.service_account import Credentials

    credentials = Credentials.from_service_account_file(
        GOOGLE_CREDENTIALS_FILE,
        scopes=["https://www.googleapis.com/auth/spreadsheets"],

    spreadsheet = client.open_by_key(GOOGLE_SHEET_ID)
    try:
        return spreadsheet.worksheet(SHEET_TITLE)
    except gspread.WorksheetNotFound:
        return spreadsheet.add_worksheet(title=SHEET_TITLE, rows=1000, cols=12)

        return
    values = [
        str(row["id"]), row["day"], row["time"], row["status"], row["service"],
        row["model"] or "", row["name"], row["phone"], row["username"] or "",
        row["created_at"] or "", row["ready_at"] or "",
    ]
    try:
        worksheet = _worksheet()
        if not worksheet.get_all_values():
            worksheet.append_row(SHEET_HEADER)
        try:
            cell = worksheet.find(str(row["id"]), in_column=1)
        except gspread.exceptions.CellNotFound:
            cell = None
        if cell is not None:
            worksheet.update(values=[values], range_name=f"A{cell.row}:K{cell.row}")
        else:
            worksheet.append_row(values)
    except Exception:
        logging.exception("Google Sheets: не удалось синхронизировать запись #%s", booking_id)


def spawn_sheet_sync(booking_id: int) -> None:
    if not sheets_enabled():
        return
    try:
        loop = asyncio.get_running_loop()
    except RuntimeError:
        return
    task = loop.create_task(asyncio.to_thread(sync_booking_to_sheet, booking_id))
    _background_tasks.add(task)
    task.add_done_callback(_background_tasks.discard)

def days_kb(prefix: str = "day", back_cb: str | None = "back:svc",
            exclude: sqlite3.Row | None = None):
    """Дни, в которых остались свободные слоты. None — если свободных дней нет."""
    builder = InlineKeyboardBuilder()
    today = now().date()
    added = 0
    for i in range(DAYS_AHEAD):
        d = today + timedelta(days=i)
        if d.weekday() in DAYS_OFF:
            continue
        day_iso = d.isoformat()
        if not free_slots(day_iso, exclude=exclude):
            continue
        if i == 0:
            label = "Сегодня"
        elif i == 1:
            label = "Завтра"
        else:
            label = f"{d:%d.%m} {WEEKDAYS[d.weekday()]}"
        builder.button(text=label, callback_data=f"{prefix}:{day_iso}")
        added += 1
    if not added:
        return None
    builder.adjust(3)
    if back_cb:
        builder.row(InlineKeyboardButton(text="⬅️ Назад", callback_data=back_cb))
    return builder.as_markup()

def cb_int(data: str | None, index: int = 1) -> int | None:
    try:
        return int((data or "").split(":")[index])
    except (IndexError, ValueError):
        return None


async def notify(bot: Bot, chat_id: int, text: str, markup=None, attempt: int = 0) -> bool:
    """Отправка с защитой от блокировок, флуд-лимитов и битого HTML."""
    try:
        await bot.send_message(
            chat_id, text, reply_markup=markup,
            parse_mode=ParseMode.HTML, link_preview_is_disabled=True,
        )
        return True
    except TelegramForbiddenError:
        return True  # пользователь заблокировал бота — повторять бессмысленно
    except TelegramRetryAfter as e:
        if attempt >= 2:
            return False
        await asyncio.sleep(min(e.retry_after, 20))
        return await notify(bot, chat_id, text, markup, attempt + 1)
    except TelegramBadRequest as e:
        logging.warning("HTML отклонён для %s: %s", chat_id, e)
        try:
            await bot.send_message(
                chat_id, re.sub(r"<[^>]+>", "", text),
                reply_markup=markup, parse_mode=None,
            )
            return True
        except TelegramAPIError:
            return False
    except TelegramAPIError:
        logging.exception("Не удалось отправить сообщение %s", chat_id)
        return False


async def notify_photo(bot: Bot, chat_id: int, photo_id: str, caption: str, markup=None) -> bool:
    try:
        await bot.send_photo(
            chat_id, photo_id, caption=caption[:1024],
            reply_markup=markup, parse_mode=ParseMode.HTML,
        )
        return True
    except TelegramAPIError:
        logging.exception("Не удалось отправить фото в %s", chat_id)
        return await notify(bot, chat_id, caption, markup)


async def send_plain(bot: Bot, chat_id: int, text: str) -> bool:
    try:
        await bot.send_message(chat_id, text, parse_mode=None)
        return True
    except TelegramForbiddenError:
        return False
    except TelegramRetryAfter as e:
        await asyncio.sleep(min(e.retry_after, 20))
        try:
            await bot.send_message(chat_id, text, parse_mode=None)
            return True
        except TelegramAPIError:
            return False
    except TelegramAPIError:
        logging.warning("Рассылка не доставлена %s", chat_id)
        return False


def my_text(user_id: int) -> tuple[str, object]:
    rows = active_bookings(user_id)
    ready = ready_bookings(user_id)
    if not rows and not ready:
        return ("У вас нет активных записей.\n\n"
                "Нажмите «📝 Записаться», чтобы выбрать время."), None
    lines: list[str] = []
    if rows:
        lines.append("<b>Ваши записи</b>")
        for row in rows:
            mark = "✅" if row["status"] == "confirmed" else "⏳"
            lines.append(
                f"{mark} <b>{fmt_day(row['day'])}, {row['time']}</b>\n"
                f"   {esc(row['service'])}, {esc(row['model'])}"
            )
    if ready:
        if lines:
            lines.append("")
        lines.append("<b>🔔 Готово к выдаче</b>")
        for row in ready:
            lines.append(
                f"• {esc(row['service'])}, {esc(row['model'])}\n"
                f"   {esc(ADDRESS)}, {esc(WORK_TEXT)}"
            )
    if rows:
        lines.append("\nПеренести или отменить — кнопками ниже.")
    return "\n".join(lines), my_kb(rows)

async def send_warranty(message: Message, raw: str) -> None:
    phone = normalize_phone(raw)
    if phone is None:
        return await message.answer("Не понял номер. Пример: +7 900 000-00-00")
    rows = warranty_rows(phone)
    if not rows:
        return await message.answer(
            "Не нашёл ремонт с таким номером. Если гарантия должна действовать — "
            f"позвоните нам: {esc(PHONE)}", parse_mode=ParseMode.HTML)
    today = now().date()
    lines = ["<b>🛡 Гарантия</b>"]
    for row in rows[:5]:
        try:
            start = date.fromisoformat((row["ready_at"] or "")[:10])
        except ValueError:
            continue
        until = add_months(start, WARRANTY_MONTHS)
        left = (until - today).days
        if left >= 0:
            lines.append(
                f"• {esc(row['service'])}, {esc(row['model'])}\n"
                f"   действует до <b>{until:%d.%m.%Y}</b> (осталось {left} дн.)")
        else:
            lines.append(
                f"• {esc(row['service'])}, {esc(row['model'])}\n"
                f"   истекла {until:%d.%m.%Y}")
    lines.append("\nЛюбые вопросы по гарантии: " + esc(PHONE))
    await message.answer("\n".join(lines), parse_mode=ParseMode.HTML)

@router.callback_query(F.data.startswith("rate:"))
async def rate_booking(call: CallbackQuery, bot: Bot):
    parts = (call.data or "").split(":")
    if len(parts) != 3 or not parts[1].isdigit() or not parts[2].isdigit():
        return await call.answer("Некорректная оценка", show_alert=True)
    booking_id, rating = int(parts[1]), int(parts[2])
    row = get_booking(booking_id)
    if row is None or row["user_id"] != call.from_user.id:
        return await call.answer("Запись не найдена", show_alert=True)
    if not 1 <= rating <= 5:
        return await call.answer("Оценка от 1 до 5", show_alert=True)
    with db() as conn:
        if conn.execute("SELECT 1 FROM reviews WHERE booking_id=?",
                        (booking_id,)).fetchone():
            return await call.answer("Вы уже оценили, спасибо!")
    save_review(booking_id, call.from_user.id, rating)
    stars = "⭐" * rating
    markup = None
    if rating >= 4:
        text = f"Спасибо! {stars}\n\nРады, что всё хорошо."
        if REVIEW_URL:
            builder = InlineKeyboardBuilder()
            builder.row(InlineKeyboardButton(text="Оставить отзыв", url=REVIEW_URL))
            markup = builder.as_markup()
            text += "\nБудем благодарны за отзыв на карте — это очень помогает."
    else:
        text = (f"{stars}\n\nСпасибо за честность. Передал мастеру — "
                "постараемся разобраться и стать лучше.")
        builder = InlineKeyboardBuilder()
        builder.row(InlineKeyboardButton(text="💬 Написать нам", url=TELEGRAM_URL))
        markup = builder.as_markup()
        if ADMIN_ID:
            await notify(bot, ADMIN_ID,
                         f"⚠️ Низкая оценка {rating}/5\n"
                         f"#{row['id']} {esc(row['name'])}, {esc(row['service'])}\n"

@router.callback_query(F.data == "confirm")
async def confirm(call: CallbackQuery, state: FSMContext, bot: Bot):
    data = await state.get_data()
    required = ("service", "model", "day", "time", "name", "phone")
    if any(not data.get(key) for key in required):
        await state.clear()
        await call.message.edit_text("Данные заявки потерялись. Начните заново: «📝 Записаться».")
        return await call.answer()

    day, slot = data["day"], data["time"]
    if not is_bookable_day(day) or slot not in slot_times() or slot not in free_slots(day):
        await state.clear()
        await call.message.edit_text("Это время уже недоступно. Выберите другое: «📝 Записаться».")
        return await call.answer()
    if slot_dt(day, slot) <= now() + timedelta(minutes=LEAD_MINUTES):
        await state.clear()
        await call.message.edit_text("Это время уже прошло. Выберите другое: «📝 Записаться».")
        return await call.answer()
    if active_count(call.from_user.id) >= MAX_ACTIVE:
        await state.clear()
        await call.message.edit_text(
            f"У вас уже {MAX_ACTIVE} активных записи. Отмените одну в «📋 Мои записи» "
            "или позвоните нам.")
        return await call.answer()

    try:
        with db() as conn:
            cursor = conn.execute(
                """INSERT INTO bookings
                   (user_id, username, name, phone, service, price, model, day, time,
                    status, created_at, photo_id)
                   VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, 'pending', ?, ?)""",
                (call.from_user.id, call.from_user.username, data["name"], data["phone"],
                 data["service"], data.get("price"), data["model"], day, slot,
                 iso_now(), data.get("photo_id")),
            )
            booking_id = int(cursor.lastrowid)
    except sqlite3.IntegrityError:
        await state.clear()
        await call.message.edit_text(
            "Это время только что заняли. Нажмите «📝 Записаться» и выберите другое.")
        return await call.answer()

    await state.clear()
    try:
        await call.message.edit_text(
            "✅ <b>Заявка принята!</b>\n\n"
            f"🗓 {fmt_day(day)} в {slot}\n"
            f"🔧 {esc(data['service'])}, {esc(data['model'])}\n\n"
            f"📍 {esc(ADDRESS)}\n📞 {esc(PHONE)}\n\n"
            "Скоро подтвердим запись — пришлём сообщение.\n"
            "Отменить или перенести можно в «📋 Мои записи».",
            parse_mode=ParseMode.HTML)
    except TelegramBadRequest:
        pass
    await call.answer("Заявка отправлена")

    if ADMIN_ID:
        text = admin_booking_text(booking_id, data, call.from_user.username)
        if data.get("photo_id"):
            await notify_photo(bot, ADMIN_ID, data["photo_id"], text,
                               admin_booking_kb(booking_id))
        else:
            await notify(bot, ADMIN_ID, text, markup=admin_booking_kb(booking_id))
    spawn_sheet_sync(booking_id)

def parse_when(raw: str) -> str | None:
    text = (raw or "").strip().lower().rstrip(".")
    if text in ("сегодня", "today"):
        return today_str()
    if text in ("завтра", "tomorrow"):
        return (now().date() + timedelta(days=1)).isoformat()
    for fmt in ("%Y-%m-%d", "%d.%m.%Y", "%d.%m"):
        try:
            value = datetime.strptime(text, fmt).date()
        except ValueError:
            continue
        if fmt == "%d.%m":
            value = value.replace(year=now().year)
            if value < now().date():
                value = value.replace(year=value.year + 1)
        return value.isoformat()
    return None

async def do_close(message: Message, state: FSMContext, close: bool) -> None:
    parts = (message.text or "").split()
    if parts and parts[0].startswith("/"):
        parts = parts[1:]
    day = parse_when(parts[0]) if parts else None
    slot = parts[1] if len(parts) > 1 else None
    if day is None or (slot is not None and slot not in slot_times()):
        await message.answer(
            "Не понял. Пример: <code>/close 20.11</code> или "
            "<code>/close завтра 14:00</code>", parse_mode=ParseMode.HTML)
        return
    if close:
        add_closure(day, slot)
        affected = [r for r in bookings_for_day(day) if slot is None or r["time"] == slot]
        text = f"🚫 Закрыто: {fmt_day(day)}" + (f" {slot}" if slot else " (весь день)")
        if affected:
            text += (f"\n\n⚠️ На это время уже есть записи ({len(affected)}). "
                     "Перенесите или отмените их:")
            for row in affected[:5]:
                text += (f"\n• {row['time']} — {esc(row['name'])}, "
                         f"{esc(fmt_phone(row['phone']))}")
    else:
        removed = remove_closure(day, slot)
        text = (f"✅ Открыто: {fmt_day(day)}" + (f" {slot}" if slot else " (весь день)")
                if removed else "Такого закрытия не было.")
    await state.set_state(None)
    await message.answer(text, parse_mode=ParseMode.HTML)

def day_report(day: str) -> str:
    rows = bookings_for_day(day)
    title = f"<b>{fmt_day(day)}</b>"
    if not rows:
        return f"{title}\nЗаписей нет."
    marks = {"confirmed": "✅", "pending": "⏳", "ready": "🔔", "done": "🏁"}
    lines = [f"{title} — {len(rows)} запис."]
    for row in rows:
        mark = marks.get(row["status"], "•")
        lines.append(
            f"{mark} <b>{row['time']}</b> — {esc(row['name'])}, {esc(fmt_phone(row['phone']))}\n"
            f"    {esc(row['service'])}, {esc(row['model'])}"
        )
    return "\n".join(lines)


def week_report() -> str:
    lines = ["<b>Ближайшие дни</b>"]
    total = 0
    for i in range(DAYS_AHEAD):
        d = now().date() + timedelta(days=i)
        if d.weekday() in DAYS_OFF:
            continue
        rows = bookings_for_day(d.isoformat())
        total += len(rows)
        lines.append(f"{fmt_day(d.isoformat())}: {len(rows)}")
    lines.append(f"\nВсего за период: {total}")
    return "\n".join(lines)


def stats_report() -> str:

        confirmed = scalar("SELECT COUNT(*) FROM bookings WHERE status='confirmed' AND day>=?", (today,))

def find_bookings(query: str) -> list[sqlite3.Row]:
    text = query.strip()
    digits = re.sub(r"\D", "", text)
    with db() as conn:
        if len(digits) >= 4:
            return conn.execute(
                "SELECT * FROM bookings WHERE phone LIKE ? OR name LIKE ? OR model LIKE ? "
                "ORDER BY day DESC, time DESC LIMIT 20",
                (f"%{digits}%", f"%{text}%", f"%{text}%"),
            ).fetchall()
        return conn.execute(
            "SELECT * FROM bookings WHERE name LIKE ? OR model LIKE ? "
            "ORDER BY day DESC, time DESC LIMIT 20",
            (f"%{text}%", f"%{text}%"),
        ).fetchall()

async def send_reminders(bot: Bot) -> None:
    current = now()

    # 1. Вечернее напоминание на завтра
    if current.hour >= REMIND_EVENING_HOUR:
        tomorrow = (current.date() + timedelta(days=1)).isoformat()
        with db() as conn:
            rows = conn.execute(
                "SELECT id, user_id, time, service, model FROM bookings "
                "WHERE day=? AND status IN ('pending','confirmed') AND reminded_day=0",
                (tomorrow,),
            ).fetchall()
        for row in rows:
            text = (
                "🔔 <b>Напоминание о записи</b>\n\n"
                f"Завтра в <b>{row['time']}</b> ждём вас.\n"
                f"🔧 {esc(row['service'])}, {esc(row['model'])}\n"
                f"📍 {esc(ADDRESS)}\n📞 {esc(PHONE)}\n\n"
                "Если планы изменились — отмените запись в «📋 Мои записи»."
            )
            if await notify(bot, row["user_id"], text):
                with db() as conn:
                    conn.execute("UPDATE bookings SET reminded_day=1 WHERE id=?", (row["id"],))
            await asyncio.sleep(0.05)

    # 2. Напоминание за REMIND_SOON_HOURS часов до визита
    with db() as conn:
        rows = conn.execute(
            "SELECT id, user_id, day, time, service, model FROM bookings "
            "WHERE day=? AND status IN ('pending','confirmed') AND reminded_soon=0",
            (current.date().isoformat(),),
        ).fetchall()
    for row in rows:
        left = slot_dt(row["day"], row["time"]) - current
        if not timedelta(0) < left <= timedelta(hours=REMIND_SOON_HOURS):
            continue
        hours = max(1, round(left.total_seconds() / 3600))
        text = (
            "⏰ <b>Скоро ждём вас</b>\n\n"
            f"Сегодня в <b>{row['time']}</b> (~{hours} ч).\n"
            f"🔧 {esc(row['service'])}, {esc(row['model'])}\n"
            f"📍 {esc(ADDRESS)}"
        )
        if await notify(bot, row["user_id"], text):
            with db() as conn:
                conn.execute("UPDATE bookings SET reminded_soon=1 WHERE id=?", (row["id"],))
        await asyncio.sleep(0.05)


def review_due_bookings() -> list[sqlite3.Row]:

async def reminder_loop(bot: Bot) -> None:
    while True:
        try:
            await send_reminders(bot)
            await send_review_requests(bot)
        except asyncio.CancelledError:
            raise
        except Exception:
            logging.exception("Сбой в цикле напоминаний")
        await asyncio.sleep(REMINDER_TICK)

def setup_logging() -> None:
    try:
        sys.stdout.reconfigure(encoding="utf-8", errors="replace")
    except (AttributeError, ValueError):
        pass
    formatter = logging.Formatter("%(asctime)s %(levelname)-8s %(name)s: %(message)s")
    root = logging.getLogger()
    root.setLevel(logging.INFO)
    stream = logging.StreamHandler(sys.stdout)
    stream.setFormatter(formatter)
    root.addHandler(stream)
    try:
        file_handler = logging.FileHandler(LOG_PATH, encoding="utf-8")
        file_handler.setFormatter(formatter)
        root.addHandler(file_handler)
    except OSError:
        logging.warning("Не удалось открыть файл логов %s", LOG_PATH)

def make_error_handler(bot: Bot):
    async def on_error(event) -> bool:
        logging.error("Ошибка при обработке апдейта", exc_info=event.exception)
        if ADMIN_ID:
            summary = f"{type(event.exception).__name__}: {event.exception}"
            await send_plain(bot, ADMIN_ID, "⚠️ Ошибка в боте:\n" + summary[:3500])
        return True

    return on_error


async def main() -> None:
    setup_logging()
    if not BOT_TOKEN or BOT_TOKEN == "ВСТАВЬТЕ_ТОКЕН_ОТ_BOTFATHER":
        logging.error("Не задан BOT_TOKEN. Укажите его в переменной окружения или в .env.")
        raise SystemExit(1)
    if not ADMIN_ID:
        logging.warning("ADMIN_ID не задан: заявки не будут приходить владельцу. "
                        "Узнайте свой ID командой /id и добавьте его в .env.")

    init_db()
    bot = make_bot()
    dispatcher = Dispatcher(storage=MemoryStorage())
    dispatcher.include_router(router)
    dispatcher.errors.register(make_error_handler(bot))
    dispatcher.startup.register(set_commands)

    await bot.delete_webhook(drop_pending_updates=True)
    reminder_task = asyncio.create_task(reminder_loop(bot))
    try:
        await dispatcher.start_polling(bot)
    finally:
        reminder_task.cancel()
        try:
            await reminder_task
        except asyncio.CancelledError:
            pass
        await bot.session.close()


if __name__ == "__main__":
    try:
        asyncio.run(main())
    except (KeyboardInterrupt, SystemExit):
        pass
